cmake_minimum_required(VERSION 3.21)
project(OpenDrop VERSION 0.3.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Network Widgets Concurrent)
add_library(opendrop_core desktop/src/protocol.cpp desktop/src/network.cpp)
target_include_directories(opendrop_core PUBLIC desktop/src)
target_link_libraries(opendrop_core PUBLIC Qt6::Core Qt6::Network Qt6::Concurrent)
qt_add_executable(opendrop desktop/src/main.cpp desktop/src/window.cpp)
target_link_libraries(opendrop PRIVATE opendrop_core Qt6::Widgets)
foreach(target opendrop_core opendrop)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /utf-8)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endforeach()
if(WIN32)
    set_target_properties(opendrop PROPERTIES WIN32_EXECUTABLE TRUE)
endif()
include(GNUInstallDirs)
install(TARGETS opendrop RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
if(UNIX AND NOT APPLE)
    install(FILES desktop/dev.opendrop.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
    install(FILES desktop/opendrop.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps)
endif()
include(CTest)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    add_executable(opendrop_tests tests/core_test.cpp desktop/src/window.cpp)
    target_link_libraries(opendrop_tests PRIVATE opendrop_core Qt6::Test Qt6::Widgets)
    if(WIN32)
        target_link_libraries(opendrop_tests PRIVATE ws2_32)
    endif()
    if(MSVC)
        target_compile_options(opendrop_tests PRIVATE /utf-8)
    endif()
    add_test(NAME core_and_ui COMMAND opendrop_tests)
    set_tests_properties(core_and_ui PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 45)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_test(NAME protocol_integration COMMAND ${Python3_EXECUTABLE}
        ${CMAKE_SOURCE_DIR}/tests/integration.py $<TARGET_FILE:opendrop>)
    set_tests_properties(protocol_integration PROPERTIES TIMEOUT 90 SKIP_RETURN_CODE 77)
endif()
